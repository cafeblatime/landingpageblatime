try { const urlParams = new URLSearchParams(window.location.search); let rawTable = urlParams.get('table') || urlParams.get('Table') || '1'; rawTable = rawTable.replace(/[^0-9]/g, '') || '1'; // Extract unique booking identifier from URL (defaults to 'latest' if missing) const sessionId = urlParams.get('t') || urlParams.get('submission_id') || 'latest'; document.getElementById('tableTitle').innerText = "Table " + rawTable + " Focus Session"; // Table & Session-Specific LocalStorage Key const storageKey = 'session_start_table_' + rawTable + '_' + sessionId; let startTime = localStorage.getItem(storageKey); if (!startTime) { startTime = Date.now(); localStorage.setItem(storageKey, startTime); } else { startTime = parseInt(startTime, 10); } // Check persistent notification button state for this specific session if (localStorage.getItem('alerts_enabled_table_' + rawTable + '_' + sessionId) === 'true') { const btn = document.getElementById('notifyBtn'); const status = document.getElementById('statusText'); if (btn) { btn.innerText = "✓ Alerts Enabled"; btn.style.background = "#34C759"; btn.disabled = true; if (status) status.innerText = "You will receive an alert 7 minutes before expiry."; } } // --- TIMER ENGINE --- const BUFFER_DURATION_MS = 1 * 60 * 1000; const MAIN_DURATION_MS = 7 * 60 * 1000; const TOTAL_DURATION_MS = BUFFER_DURATION_MS + MAIN_DURATION_MS; const bufferTimerEl = document.getElementById('bufferTimer'); const mainTimerEl = document.getElementById('mainTimer'); const chimeSound = document.getElementById('chimeSound'); let alertPlayed = false; const interval = setInterval(() => { const elapsedMs = Date.now() - startTime; if (elapsedMs < BUFFER_DURATION_MS) { let remainingBufferMs = BUFFER_DURATION_MS - elapsedMs; let bufferSecs = Math.floor(remainingBufferMs / 1000); let bMins = Math.floor(bufferSecs / 60); let bSecs = bufferSecs % 60; bufferTimerEl.innerText = `${bMins < 10 ? '0' : ''}${bMins}:${bSecs < 10 ? '0' : ''}${bSecs}`; bufferTimerEl.classList.add('warning'); mainTimerEl.innerText = "45:00"; } else if (elapsedMs < TOTAL_DURATION_MS) { bufferTimerEl.innerText = "00:00"; bufferTimerEl.classList.remove('warning'); bufferTimerEl.classList.add('expired'); let remainingMainMs = TOTAL_DURATION_MS - elapsedMs; let mainSecs = Math.floor(remainingMainMs / 1000); let mMins = Math.floor(mainSecs / 60); let mSecs = mainSecs % 60; mainTimerEl.innerText = `${mMins}:${mSecs < 10 ? '0' : ''}${mSecs}`; if (mainSecs <= 420 && !alertPlayed) { mainTimerEl.classList.add('warning'); chimeSound.play().catch(e => console.log('Autoplay blocked')); alertPlayed = true; } } else { bufferTimerEl.innerText = "00:00"; mainTimerEl.innerText = "0:00"; mainTimerEl.classList.remove('warning'); mainTimerEl.classList.add('expired'); clearInterval(interval); } }, 1000); } catch (err) { console.error("Initialization Error:", err); }
