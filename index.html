const SPREADSHEET_ID = SpreadsheetApp.getActiveSpreadsheet().getId(); function doPost(e) { try { const contents = JSON.parse(e.postData.contents); // Handle direct clear action from frontend if (contents.action === "clear") { const ss = SpreadsheetApp.getActiveSpreadsheet(); const liveSheet = ss.getSheetByName("Live Dashboard"); const backlogSheet = ss.getSheetByName("Backlog"); const tableNum = contents.table; const clientToken = contents.client || ""; const data = liveSheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(tableNum)) { // Explicitly writes "CLEARED" to the status column (Column B / index 2) liveSheet.getRange(i + 1, 2).setValue("CLEARED"); // Clear out the remaining tracking cells (EndTime, Token, IsPaid) liveSheet.getRange(i + 1, 3, 1, 3).setValues([["", "", false]]); break; } } if (backlogSheet) { backlogSheet.appendRow([ new Date(), "SESSION_ENDED", tableNum, "", 0, "NO", "CLEARED", "", clientToken ]); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } const payload = contents.data; let tableNum = ""; let itemsArr = []; let totalAmount = 0; let focusPass = false; let userToken = ""; if (payload && payload.fields) { payload.fields.forEach(field => { const label = (field.label || "").toLowerCase(); if (label.includes("table")) tableNum = String(field.value); if (label.includes("token") || label.includes("user")) userToken = String(field.value); if (field.type === "MULTIPLE_CHOICE" || field.type === "CHECKBOXES") { if (field.value) { const val = Array.isArray(field.value) ? field.value : [field.value]; itemsArr = itemsArr.concat(val); } } }); } itemsArr.forEach(item => { if (String(item).toLowerCase().includes("focus pass")) focusPass = true; }); const backlogSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Backlog"); backlogSheet.appendRow([ new Date(), contents.eventId || e.parameter.eventId || "WEBHOOK", tableNum, itemsArr.join(", "), totalAmount, focusPass ? "YES" : "NO", "", "", "", "", userToken ]); updateLiveDashboard(tableNum, focusPass, userToken); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { const action = e.parameter.action; liveSheet.getRange(i + 1, 3, 1, 3).setValues([["", "", false]]); break; } } if (backlogSheet) { backlogSheet.appendRow([ new Date(), "SESSION_ENDED", table, "", 0, "NO", "CLEARED", "", clientToken ]); } return replyJSON({ status: "success" }); } if (action === "checkUser") { const liveSheet = ss.getSheetByName("Live Dashboard"); const data = liveSheet.getDataRange().getValues(); let occupied = false; for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(table)) { const status = String(data[i][1]).toLowerCase(); const activeToken = String(data[i][3] || ""); if ((status === "active" || status === "expiring") && activeToken && activeToken !== clientToken) { occupied = true; } break; } } return replyJSON({ status: "success", occupied: occupied }); } if (action === "client") { const liveSheet = ss.getSheetByName("Live Dashboard"); const data = liveSheet.getDataRange().getValues(); let tableInfo = { status: "Open", endTime: null }; const targetTable = String(table || "").trim(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === targetTable) { tableInfo = { status: data[i][1], endTime: data[i][2], userToken: data[i][3] }; break; } } return replyJSON({ status: "success", tables: { [targetTable]: tableInfo }, table: tableInfo }); } if (action === "manager") { const liveSheet = ss.getSheetByName("Live Dashboard"); const backlogSheet = ss.getSheetByName("Backlog"); const liveData = liveSheet.getDataRange().getValues(); const backlogData = backlogSheet.getDataRange().getValues(); let tablesObj = {}; for (let i = 1; i < liveData.length; i++) { const tNum = String(liveData[i][0]); if (!tNum) continue; let orders = []; let totalSum = 0; let intervalCount = 0; for (let j = 1; j < backlogData.length; j++) { if (String(backlogData[j][2]) === tNum) { if (backlogData[j][3]) orders.push(backlogData[j][3]); totalSum += Number(backlogData[j][4] || 0); if (String(backlogData[j][5]).toUpperCase() === "YES") intervalCount++; } } tablesObj[tNum] = { table: tNum, status: liveData[i][1] || "Open", endTime: liveData[i][2] || null, isPaid: liveData[i][4] === true || String(liveData[i][4]).toUpperCase() === "TRUE", orders: orders, totalSum: totalSum, intervalCount: intervalCount }; } return replyJSON({ status: "success", tables: tablesObj }); } if (action === "togglePaid") { const liveSheet = ss.getSheetByName("Live Dashboard"); const data = liveSheet.getDataRange().getValues(); let newState = false; for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(table)) { const currentPaid = data[i][4] === true || String(data[i][4]).toUpperCase() === "TRUE"; newState = !currentPaid; liveSheet.getRange(i + 1, 5).setValue(newState); break; } } return replyJSON({ status: "success", isPaid: newState }); } if (action === "logReminder") { const tableNum = e.parameter.table; const clientToken = e.parameter.client; const backlogSheet = ss.getSheetByName("Backlog"); if (!backlogSheet) { ss.insertSheet("Backlog"); } backlogSheet.appendRow([ new Date(), "REMINDER_WEBHOOK", tableNum, "7-Min Reminder Requested", 0, "NO", "1", "SENT", clientToken ]); return replyJSON({ status: "success", message: "Reminder logged" }); } return replyJSON({ status: "error", message: "Invalid action" }); } function updateLiveDashboard(tableNum, focusPass, userToken) { const liveSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Live Dashboard"); const data = liveSheet.getDataRange().getValues(); let found = false; const now = new Date(); const endTime = new Date(now.getTime() + 47 * 60 * 1000); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(tableNum)) { liveSheet.getRange(i + 1, 2, 1, 4).setValues([["Active", endTime.toISOString(), userToken, false]]); found = true; break; } } if (!found) { liveSheet.appendRow([tableNum, "Active", endTime.toISOString(), userToken, false]); } } function replyJSON(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
